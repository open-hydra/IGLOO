# The I/O probe of the timing harness (tests/bench/monolith/README.md): io-probe, and the four builds of io-probe-wr
# that place -assume buffered_io on the main program, on the module holding the OPEN, on both or nowhere. A
# measurement tool, never registered with ctest; configured only by the top-level IGLOO_IO_PROBE option.
if(NOT USE_OPENMP)
  message(FATAL_ERROR "IGLOO_IO_PROBE needs USE_OPENMP=ON")
endif()
enable_language(C)

add_library(io_probe_c STATIC io_probe_c.c)
# icx's default fast model computes fma() as a rounded product plus a sum
if(CMAKE_C_COMPILER_ID MATCHES "^Intel")
  target_compile_options(io_probe_c PRIVATE -fp-model=precise)
endif()

add_executable(io-probe io_probe.f90 io_probe_kernels.f90)
target_link_libraries(io-probe PRIVATE ${lib} ORION io_probe_c)
target_compile_options(io-probe PRIVATE ${OpenMP_Fortran_FLAGS})
target_link_options(io-probe PRIVATE ${OpenMP_Fortran_FLAGS})
# the exact fast conversion needs IEEE multiplies and divides (no reciprocal, no reassociation)
if(CMAKE_Fortran_COMPILER_ID MATCHES "^Intel")
  set_source_files_properties(io_probe_kernels.f90 PROPERTIES COMPILE_OPTIONS "-fp-model;precise")
endif()
set_target_properties(io-probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin/ LINKER_LANGUAGE Fortran)

if(CMAKE_Fortran_COMPILER_ID MATCHES "^Intel")
  # ORION's Lib_Tecplot copied under another module name, so that the flag reaches only the copy's OPEN
  file(READ "${orion_SOURCE_DIR}/src/fortran/lib/Lib_Tecplot.f90" _tec)
  string(REPLACE "module Lib_Tecplot" "module Lib_Tecplot_bufio" _tec "${_tec}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/Lib_Tecplot_bufio.f90" "${_tec}")
  add_library(io_probe_tec_bufio STATIC "${CMAKE_CURRENT_BINARY_DIR}/Lib_Tecplot_bufio.f90")
  target_link_libraries(io_probe_tec_bufio PRIVATE ORION)
  target_compile_options(io_probe_tec_bufio PRIVATE "SHELL:-assume buffered_io")
  foreach(_v plain main mod both)
    if(_v STREQUAL "plain")
      set(_exe io-probe-wr)
    else()
      set(_exe io-probe-wr-${_v})
    endif()
    add_executable(${_exe} io_probe_write.f90)
    target_link_libraries(${_exe} PRIVATE ORION io_probe_c)
    if(_v STREQUAL "mod" OR _v STREQUAL "both")
      target_compile_definitions(${_exe} PRIVATE PROBE_TEC_COPY)
      target_link_libraries(${_exe} PRIVATE io_probe_tec_bufio)
    endif()
    if(_v STREQUAL "main" OR _v STREQUAL "both")
      target_compile_options(${_exe} PRIVATE "SHELL:-assume buffered_io")
    endif()
    set_target_properties(${_exe} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin/ LINKER_LANGUAGE Fortran)
  endforeach()
endif()
